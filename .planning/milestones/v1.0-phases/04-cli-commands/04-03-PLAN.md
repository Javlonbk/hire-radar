```typescript
import { Command } from 'commander';
import { openDatabase } from '../db/client.js';
import { queryVacancies } from '../db/vacancies.js';
import type { VacancyFilters, VacancyRow } from '../db/vacancies.js';
import type Database from 'better-sqlite3';

export interface ListDeps {
  db: Database.Database;
  out: (line: string) => void;   // stdout
  err: (line: string) => void;   // stderr
}

interface ListOptions {
  source?: string; keyword?: string; since?: string; until?: string; limit?: string;
}

export function runList(deps: ListDeps, opts: ListOptions): void {
  const filters: VacancyFilters = {
    source: opts.source,
    keyword: opts.keyword,
    since: opts.since,
    until: opts.until,
    limit: opts.limit !== undefined ? Number(opts.limit) : 20,
  };
  const rows = queryVacancies(deps.db, filters);
  if (rows.length === 0) {
    deps.err('No vacancies match');
    return;
  }
  for (const r of rows) deps.out(formatRow(r));
}
```

Implement `formatRow(r: VacancyRow): string`:
   - Build a single compact aligned line. Truncate each field with a helper `trunc(s, n)` that returns `s` if `s.length <= n` else `s.slice(0, n - 1) + '…'`. Suggested widths: title 30, company 20, location 16, source 14.
   - salary cell: if salary_min/max present → `${min ?? '?'}-${max ?? '?'} ${currency ?? ''}` else `—`.
   - Include remote_type after location (e.g. `(remote)` when set).
   - date cell: `r.date.slice(0, 10)` (YYYY-MM-DD).
   - Order: title — company — location/remote — salary — source — date. Use `padEnd` on each truncated field for column alignment, joined with `  `.

}

import { Command } from 'commander';
import { openDatabase } from '../db/client.js';
import { queryVacancies } from '../db/vacancies.js';
import type { VacancyFilters } from '../db/vacancies.js';
import type Database from 'better-sqlite3';

export interface ExportDeps {
  db: Database.Database;
  out: (text: string) => void;   // stdout — JSON only
}

interface ExportOptions {
  source?: string; keyword?: string; since?: string; until?: string; limit?: string;
}

export function runExport(deps: ExportDeps, opts: ExportOptions): void {
  const filters: VacancyFilters = {
    source: opts.source,
    keyword: opts.keyword,
    since: opts.since,
    until: opts.until,
    limit: opts.limit !== undefined ? Number(opts.limit) : undefined,   // NO default
  };
  const rows = queryVacancies(deps.db, filters);
  deps.out(JSON.stringify(rows, null, 2));   // [] when empty (JSON.stringify([]) === '[]')
}
